1. C Introduction - ការណែនាំ C

1.1 អ្វីជា C?

C គឺជាភាសា procedural ដែលបង្កើតឡើងដោយ Dennis Ritchie ក្នុងឆ្នាំ 1972 នៅ Bell Labs។

📘 ការពន្យល់៖: C គឺជាភាសា middle-level ដែលរួមបញ្ចូលទាំង high-level និង low-level។ វាជាមូលដ្ឋាននៃភាសា C++, Java, C#, PHP, JavaScript, Python។ ប្រើសម្រាប់៖
  • Operating Systems — Linux, Windows kernel
  • Embedded Systems — Arduino, IoT
  • Compilers — Python, PHP interpreters
  • Databases — MySQL, SQLite

1.2 ការដំឡើង

  • Compiler: GCC, Clang, MSVC
  • IDE: VS Code, Code::Blocks, Dev-C++
  • Online: replit.com, onlinegdb.com
# ពិនិត្យ GCC gcc --version # Compile និង Run gcc hello.c -o hello ./hello

1.3 កម្មវិធីដំបូង

#include <stdio.h> int main() { printf("Hello, C!\n"); return 0; }
📘 ការពន្យល់លម្អិត៖:
  • #include <stdio.h> — បញ្ចូល library សម្រាប់ input/output
  • int main() — ចំណុចចាប់ផ្តើមរបស់កម្មវិធី
  • printf() — បង្ហាញអត្ថបទ
  • \n — បន្ទាត់ថ្មី (newline)
  • return 0; — បញ្ជាក់ការបញ្ចប់ដោយជោគជ័យ

2. Variables & Data Types

2.1 Variables

int age = 20; float price = 9.99; char grade = 'A'; double pi = 3.14159;
📘 ការពន្យល់៖: C ជាភាសា statically-typed — ត្រូវប្រកាស type ជានិច្ច។ Variable ត្រូវតែ initialize មុនប្រើ (មិនដូច Java ដែល auto-init)។

3. Input / Output

3.1 printf()

printf("Hello, World!\n"); int age = 20; printf("Age: %d\n", age); float price = 9.99; printf("Price: %.2f\n", price); // 9.99

3.2 scanf()

int age; printf("Enter age: "); scanf("%d", &age); // ប្រើ & char name[50]; scanf("%s", name); // Array មិនត្រូវការ &
📘 ការពន្យល់៖: scanf() ត្រូវការ & (address-of) សម្រាប់ variable ធម្មតា ដើម្បីដឹងទីតាំងក្នុង memory។ Array គឺជា pointer ស្រាប់ ដូច្នេះមិនត្រូវការ &។

3.3 getchar / putchar

char c = getchar(); // អាន 1 តួអក្សរ putchar(c); // បង្ហាញ 1 តួអក្សរ

4. Operators

4.1 Arithmetic

int sum = 5 + 3; // 8 int mod = 10 % 3; // 1 sum++; // increment sum--; // decrement

4.2 Comparison

== != > < >= <=

4.3 Logical

&& // AND || // OR ! // NOT

4.4 Bitwise

& // AND | // OR ^ // XOR ~ // NOT << // Left shift >> // Right shift // ឧទាហរណ៍ 5 & 3 // 1 (0101 & 0011 = 0001) 5 | 3 // 7 5 << 1 // 10 (×2)
📘 ការពន្យល់៖: Bitwise operators ធ្វើការលើ bits។ << 1 ស្មើនឹងគុណនឹង 2, >> 1 ស្មើនឹងចែកនឹង 2។ ល្អសម្រាប់ performance ខ្ពស់។

5. Control Flow

5.1 if / else

if (age >= 18) { printf("Adult\n"); } else if (age >= 13) { printf("Teen\n"); } else { printf("Child\n"); }

5.2 switch

switch (day) { case 1: printf("Monday\n"); break; case 2: printf("Tuesday\n"); break; default: printf("Unknown\n"); }

5.3 Loops

// for loop for (int i = 0; i < 5; i++) { printf("%d ", i); } // while loop int j = 0; while (j < 5) { j++; } // do-while do { j++; } while (j < 10);

5.4 break / continue

for (int i = 0; i < 10; i++) { if (i == 3) continue; if (i == 7) break; printf("%d ", i); }

6. Functions

6.1 Basics

// Function declaration void greet() { printf("Hello!\n"); } // Call greet();

6.2 Parameters

// Function definition int add(int a, int b) { return a + b; } int result = add(2, 3); // 5

6.3 Return Values

float average(int a, int b) { return (a + b) / 2.0; }

6.4 Recursion

int factorial(int n) { if (n <= 1) return 1; return n * factorial(n - 1); } factorial(5); // 120
📘 ការពន្យល់៖: Recursion គឺ function ដែលហៅខ្លួនឯង។ ត្រូវមាន base case ដើម្បីបញ្ឈប់។ ប្រើសម្រាប់ algorithm ដូច factorial, Fibonacci, tree traversal។

7. Arrays

7.1 Basics

int nums[5] = {1, 2, 3, 4, 5}; printf("%d\n", nums[0]); // 1 // Loop តាម array for (int i = 0; i < 5; i++) { printf("%d ", nums[i]); }

7.2 Multidimensional

int matrix[2][3] = { {1, 2, 3}, {4, 5, 6} }; printf("%d\n", matrix[1][0]); // 4

7.3 Strings

char name[50] = "Dara"; char greeting[] = "Hello"; printf("%s\n", name); // Dara
📘 ការពន្យល់៖: C គ្មាន string type ទេ — ប្រើ array នៃ char ដែលបញ្ចប់ដោយ '\0' (null terminator)។

7.4 String Functions

#include <string.h> char dest[50]; strcpy(dest, "Hello"); // Copy strlen("Hello"); // 5 strcmp("a", "b"); // < 0 strcat(dest, " World"); // Concatenate

8. Pointers

8.1 Basics

int x = 10; int *ptr = &x; // ptr ផ្ទុក address របស់ x printf("%d\n", *ptr); // 10 (dereference) printf("%p\n", ptr); // address

8.2 Pointer & Array

int arr[3] = {1, 2, 3}; int *p = arr; // arr គឺ pointer ទៅ element ដំបូង printf("%d\n", *p); // 1 printf("%d\n", *(p + 1)); // 2

8.3 Pointer & Function

// Pass by reference void increment(int *n) { (*n)++; } int x = 5; increment(&x); // x ក្លាយជា 6

8.4 NULL Pointer

int *ptr = NULL; if (ptr != NULL) { printf("%d\n", *ptr); }
📘 ការពន្យល់៖: Pointer ជាចំណុចសំខាន់បំផុតក្នុង C។ វាផ្ទុក memory address ហើយអនុញ្ញាតឲ្យកែតម្លៃដោយផ្ទាល់ និង pass by reference។

9. Structures & Unions

9.1 Struct

struct Student { char name[50]; int age; float gpa; }; struct Student s1 = {"Dara", 20, 3.5}; printf("%s\n", s1.name);

9.2 Union

union Data { int i; float f; char c; }; union Data d; d.i = 10;
📘 ការពន្យល់៖: struct ផ្ទុក members ដោយឡែកពីគ្នា។ union ផ្ទុក members នៅកន្លែងតែមួយ (ប្រើម្តងមួយ) — សន្សំ memory។

9.3 typedef

typedef struct { char name[50]; int age; } Person; Person p1 = {"Dara", 20}; // មិនចាំបាច់សរសេរ struct

9.4 enum

enum Day { MON, TUE, WED, THU, FRI, SAT, SUN }; enum Day today = WED; printf("%d\n", today); // 2

10. Dynamic Memory

10.1 malloc()

#include <stdlib.h> int *ptr = (int*) malloc(5 * sizeof(int)); if (ptr == NULL) { printf("Memory allocation failed\n"); }

10.2 calloc / realloc

// calloc — ចាប់ផ្តើមដោយ 0 int *arr = (int*) calloc(5, sizeof(int)); // realloc — ប្តូរទំហំ arr = (int*) realloc(arr, 10 * sizeof(int));

10.3 free()

free(ptr); // បញ្ចេញ memory ptr = NULL; // ការពារ dangling pointer
📘 ការពន្យល់៖: C គ្មាន garbage collector។ ត្រូវហៅ free() ដោយខ្លួនឯង ដើម្បីការពារ memory leak។ បើភ្លេច នឹងធ្វើឲ្យកម្មវិធីប្រើ memory ច្រើនឡើងៗ។

11. File I/O

11.1 fopen / fclose

#include <stdio.h> FILE *fp = fopen("data.txt", "w"); if (fp == NULL) { printf("Cannot open file\n"); return 1; } fclose(fp);

11.2 Read / Write

// សរសេរ fprintf(fp, "Hello File!\n"); fputs("Hello\n", fp); fputc('A', fp); // អាន char buffer[100]; fgets(buffer, 100, fp); fscanf(fp, "%s", buffer);

11.3 File Modes

Modeនិយមន័យ

"w"សរសេរ (លុបចាស់)
"a"បន្ថែម (append)
"r+"អាន + សរសេរ
"rb"អាន binary

12. Preprocessor

12.1 #define

#define PI 3.14159 #define MAX 100 printf("%f\n", PI);

12.2 Macros

#define SQUARE(x) ((x) * (x)) #define MAX(a, b) ((a) > (b) ? (a) : (b)) printf("%d\n", SQUARE(5)); // 25

12.3 Header Files

// mylib.h #ifndef MYLIB_H #define MYLIB_H int add(int a, int b); #endif
📘 ការពន្យល់៖: Macros គឺការជំនួសអត្ថបទដោយ preprocessor មុនពេល compile។ ប្រើ () ជុំវិញ parameters ដើម្បីការពារ bug។

13. Error Handling

13.1 errno

#include <errno.h> FILE *fp = fopen("nonexistent.txt", "r"); if (fp == NULL) { printf("Error code: %d\n", errno); }

13.2 perror

perror("File error"); // Output: File error: No such file or directory
📘 ការពន្យល់៖: C គ្មាន try/catch ដូច Java។ ត្រូវពិនិត្យ return values (NULL, -1) និងប្រើ errno ឬ perror() ដើម្បីបង្ហាញ error message។

14. Best Practices

  • ប្រើ const នៅពេលអាចធ្វើបាន
  • Initialize variable ជានិច្ច
  • ពិនិត្យ NULL មុនប្រើ pointer
  • ប្រើ sizeof ជំនួស hardcode size
  • Free memory ជានិច្ច
  • ប្រើ fgets() ជំនួស gets() (safe)
  • បិទ file ជានិច្ច
  • រក្សា function តូច
  • Comment code ឲ្យច្បាស់
📘 ការពន្យល់៖: C មាន flexibility ខ្ពស់ ប៉ុន្តែត្រូវការការយកចិត្តទុកដាក់ខ្ពស់។ Bug តូចៗអាចបណ្តាលឲ្យ segmentation fault ឬ memory leak។

15. C Project

សាកល្បងបង្កើត Student Manager តូចមួយដោយប្រើ struct និង file I/O៖

#include <stdio.h> #include <string.h> #define MAX 100 typedef struct { char name[50]; int age; float gpa; } Student; int main() { Student students[MAX]; int count = 0; // បន្ថែម 3 នាក់ strcpy(students[0].name, "Dara"); students[0].age = 20; students[0].gpa = 3.5; strcpy(students[1].name, "Sok"); students[1].age = 22; students[1].gpa = 3.8; strcpy(students[2].name, "Anna"); students[2].age = 19; students[2].gpa = 3.9; count = 3; // បង្ហាញ printf("=== Student List ===\n"); for (int i = 0; i < count; i++) { printf("%s (Age: %d, GPA: %.1f)\n", students[i].name, students[i].age, students[i].gpa); } // Save to file FILE *fp = fopen("students.txt", "w"); if (fp) { for (int i = 0; i < count; i++) { fprintf(fp, "%s %d %.1f\n", students[i].name, students[i].age, students[i].gpa); } fclose(fp); printf("\nSaved to students.txt\n"); } return 0; }
📘 ការពន្យល់៖: គម្រោងនេះប្រើ struct, array, string functions និង file I/O។ សាកល្បងបន្ថែម៖ menu, add/delete student, ឬ search តាមឈ្មោះ។
គន្លឹះ៖: C ជាភាសាដ៏សំខាន់សម្រាប់អ្នកចង់យល់ពី low-level programming។ សាកល្បងគម្រោងតូចៗ — Calculator, Bank System, ឬ Game តូចៗ។
