1. MySQL Introduction - ការណែនាំ MySQL

1.1 អ្វីជា MySQL?

MySQL គឺជា Relational Database Management System (RDBMS) ដែលប្រើភាសា SQL សម្រាប់គ្រប់គ្រងទិន្នន័យ។

📘 ការពន្យល់៖ MySQL គឺជាប្រព័ន្ធគ្រប់គ្រង database ដែលពេញនិយមបំផុតមួយក្នុងពិភពលោក។ វាជា open-source និងឥតគិតថ្លៃ។ ប្រើសម្រាប់៖
  • WordPress — គេហទំព័ររាប់លាន
  • Facebook — ផ្ទុកទិន្នន័យ user
  • YouTube — metadata វីដេអូ
  • Shopify, Twitter ជាដើម

1.2 RDBMS

RDBMS គឺជាប្រព័ន្ធដែលរក្សាទិន្នន័យជា តារាង (Tables) ដែលមានទំនាក់ទំនងគ្នា។

📘 ការពន្យល់៖: RDBMS ប្រើ relations (តារាង) ដើម្បីផ្ទុកទិន្នន័យ។ ឧទាហរណ៍ តារាង users និង orders អាចភ្ជាប់គ្នាតាម user_id។

1.3 ការដំឡើង

  • XAMPP / MAMP — ងាយស្រួលបំផុត
  • MySQL Community Server — ដំឡើងដោយផ្ទាល់
  • MySQL Workbench — GUI tool
  • phpMyAdmin — Web-based interface

1.4 ការភ្ជាប់

# Terminal mysql -u root -p # ក្នុង XAMPP ចូល phpMyAdmin http://localhost/phpmyadmin
📘 ការពន្យល់៖: ប្រើ mysql -u root -p ដើម្បីចូល CLI។ បញ្ចូល password រួចចាប់ផ្តើមបញ្ជា SQL។ ឬប្រើ GUI tool ដូចជា phpMyAdmin ឬ MySQL Workbench។

2. Databases & Tables

2.1 CREATE DATABASE

CREATE DATABASE mydb; -- បង្កើតបើមិនទាន់មាន CREATE DATABASE IF NOT EXISTS mydb;

2.2 USE / DROP

USE mydb; DROP DATABASE mydb; -- លុបទាំងស្រុង
📘 ការពន្យល់៖: USE ជ្រើសរើស database បច្ចុប្បន្ន។ DROP DATABASE លុបទាំង database និង tables ទាំងអស់ — ប្រុងប្រយ័ត្ន!

2.3 CREATE TABLE

CREATE TABLE users ( id INT AUTO_INCREMENT PRIMARY KEY, name VARCHAR(50) NOT NULL, email VARCHAR(100) UNIQUE, age INT, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP );

2.4 ALTER TABLE

ALTER TABLE users ADD phone VARCHAR(20); ALTER TABLE users DROP COLUMN phone; ALTER TABLE users MODIFY name VARCHAR(100);
📘 ការពន្យល់៖: AUTO_INCREMENT បង្កើត id ស្វ័យប្រវត្តិ។ PRIMARY KEY ជាកូនសោចម្បងដែលមិនអាចស្ទួន។ ALTER TABLE ប្រើសម្រាប់កែរចនាសម្ព័ន្ធ table។

3. Data Types

3.1 Numeric

  • INT — ចំនួនគត់ (4 bytes)
  • BIGINT — ចំនួនធំ (8 bytes)
  • DECIMAL(10,2) — ទសភាគច្បាស់លាស់ (សម្រាប់លុយ)
  • FLOAT, DOUBLE — ទសភាគប្រហែល

3.2 String

  • VARCHAR(255) — អត្ថបទអថេរ (ពេញនិយម)
  • CHAR(10) — អត្ថបទថេរ
  • TEXT — អត្ថបទវែង
  • BLOB — ទិន្នន័យ binary

3.3 Date & Time

  • DATE — YYYY-MM-DD
  • TIME — HH:MM:SS
  • DATETIME — ទាំង date និង time
  • TIMESTAMP — ជាមួយ timezone
  • YEAR — ឆ្នាំ

3.4 ENUM / JSON

status ENUM('active', 'inactive', 'banned') settings JSON
📘 ការពន្យល់៖: ជ្រើសរើស type ឲ្យសមស្រប — DECIMAL សម្រាប់លុយ (កុំប្រើ FLOAT)។ VARCHAR សម្រាប់អត្ថបទខ្លី។ ENUM សម្រាប់តម្លៃជាក់លាក់។

4. CRUD Operations

4.1 INSERT

INSERT INTO users (name, email, age) VALUES ('Dara', 'dara@example.com', 20); -- ច្រើនជួរក្នុងពេលតែមួយ INSERT INTO users (name, age) VALUES ('Sok', 25), ('Anna', 22);

4.2 SELECT

SELECT * FROM users; SELECT name, age FROM users;

4.3 UPDATE

UPDATE users SET age = 21 WHERE name = 'Dara';

4.4 DELETE

DELETE FROM users WHERE age < 18; -- លុបទាំងអស់ (ប្រុងប្រយ័ត្ន!) TRUNCATE TABLE users;
📘 ការពន្យល់៖: INSERT បន្ថែម, SELECT អាន, UPDATE កែ, DELETE លុប។ ត្រូវប្រើ WHERE ជាមួយ UPDATE/DELETE ជានិច្ច — បើមិនដូច្នេះ វានឹងប៉ះពាល់គ្រប់ជួរ!

5. WHERE Clause

5.1 WHERE Basics

SELECT * FROM users WHERE age = 20; SELECT * FROM users WHERE age > 18; SELECT * FROM users WHERE age != 20;

5.2 AND / OR

SELECT * FROM users WHERE age > 18 AND age < 60; SELECT * FROM users WHERE name = 'Dara' OR name = 'Sok';

5.3 IN / BETWEEN

SELECT * FROM users WHERE age IN (18, 20, 25); SELECT * FROM users WHERE age BETWEEN 18 AND 30;

5.4 LIKE

-- ចាប់ផ្តើមដោយ D SELECT * FROM users WHERE name LIKE 'D%'; -- មាន "ar" នៅក្នុង SELECT * FROM users WHERE name LIKE '%ar%';
📘 ការពន្យល់៖: % តំណាងឲ្យ 0 ឬច្រើនតួអក្សរ។ _ តំណាងឲ្យ 1 តួអក្សរ។ IN ជាជម្រើសនៃតម្លៃច្រើន។ BETWEEN ជាជួរតម្លៃ។

6. Sorting & Limiting

6.1 ORDER BY

SELECT * FROM users ORDER BY age ASC; -- តូចទៅធំ SELECT * FROM users ORDER BY age DESC; -- ធំទៅតូច SELECT * FROM users ORDER BY age DESC, name ASC;

6.2 LIMIT

SELECT * FROM users LIMIT 10; SELECT * FROM users LIMIT 10 OFFSET 20; -- ទំព័រទី 3

6.3 DISTINCT

SELECT DISTINCT age FROM users;
📘 ការពន្យល់៖: ORDER BY តម្រៀបលទ្ធផល។ LIMIT កំណត់ចំនួនជួរ (ល្អសម្រាប់ pagination)។ DISTINCT លុបតម្លៃស្ទួន។

7. Functions

7.1 Aggregate

SELECT COUNT(*) FROM users; -- ចំនួន SELECT SUM(price) FROM orders; -- ផលបូក SELECT AVG(age) FROM users; -- មធ្យមភាគ SELECT MAX(age), MIN(age) FROM users; -- អតិបរមា/អប្បបរមា

7.2 String

CONCAT(first, ' ', last) UPPER(name), LOWER(name) LENGTH(name) SUBSTRING(name, 1, 3) TRIM(name) REPLACE(name, 'a', 'A')

7.3 Date

NOW(), CURDATE(), CURTIME() YEAR(created_at), MONTH(created_at) DATEDIFF(end_date, start_date) DATE_FORMAT(created_at, '%Y-%m-%d')

7.4 NULL Handling

SELECT IFNULL(phone, 'N/A') FROM users; SELECT COALESCE(phone, email, 'N/A') FROM users;
📘 ការពន្យល់៖: IFNULL ជំនួស NULL ដោយតម្លៃផ្សេង។ COALESCE return តម្លៃមិន NULL ដំបូង។ មានប្រយោជន៍ពេលបង្ហាញទិន្នន័យ។

8. Group By & Having

8.1 GROUP BY

SELECT age, COUNT(*) AS total FROM users GROUP BY age;

8.2 HAVING

SELECT age, COUNT(*) AS total FROM users GROUP BY age HAVING total > 5;
📘 ការពន្យល់៖: GROUP BY ប្រមូលជួរជាក្រុម (ឧ. តាមអាយុ)។ HAVING ជា WHERE សម្រាប់ក្រុម — ត្រូវប្រើបន្ទាប់ពី GROUP BY។

9. Joins

9.1 INNER JOIN

SELECT u.name, o.total FROM users u INNER JOIN orders o ON u.id = o.user_id;

9.2 LEFT / RIGHT JOIN

-- យក user ទាំងអស់ ទោះគ្មាន order SELECT u.name, o.total FROM users u LEFT JOIN orders o ON u.id = o.user_id;

9.3 CROSS JOIN

SELECT * FROM colors CROSS JOIN sizes;

9.4 SELF JOIN

SELECT a.name AS employee, b.name AS manager FROM employees a JOIN employees b ON a.manager_id = b.id;
📘 ការពន្យល់៖: INNER JOIN យកតែជួរដែលត្រូវគ្នាទាំងពីរ table។ LEFT JOIN យកទាំងអស់ពី table ឆ្វេង ទោះគ្មានក្នុង table ស្តាំ។ SELF JOIN ភ្ជាប់ table ខ្លួនឯង (ដូច employee-manager)។

10. Subqueries

10.1 Basics

SELECT * FROM users WHERE age > (SELECT AVG(age) FROM users);

10.2 IN / EXISTS

SELECT * FROM users WHERE id IN (SELECT user_id FROM orders); SELECT * FROM users u WHERE EXISTS (SELECT 1 FROM orders WHERE user_id = u.id);
📘 ការពន្យល់៖: Subquery ជា query ក្នុង query។ ប្រើ IN នៅពេល return តម្លៃច្រើន, EXISTS នៅពេលគ្រាន់តែចង់ដឹងថាមានឬអត់។

11. Keys & Constraints

11.1 PRIMARY KEY

CREATE TABLE users ( id INT AUTO_INCREMENT PRIMARY KEY, name VARCHAR(50) );

11.2 FOREIGN KEY

CREATE TABLE orders ( id INT PRIMARY KEY AUTO_INCREMENT, user_id INT, FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE );

11.3 UNIQUE

email VARCHAR(100) UNIQUE

11.4 NOT NULL / DEFAULT

name VARCHAR(50) NOT NULL, status VARCHAR(20) DEFAULT 'active'
📘 ការពន្យល់៖: PRIMARY KEY សម្គាល់ជួរនីមួយៗ (unique + not null)។ FOREIGN KEY ភ្ជាប់ទៅ table ផ្សេង (referential integrity)។ ON DELETE CASCADE លុបជួរកូន ពេលលុបជួរមេ។

12. Indexes & Views

12.1 Indexes

CREATE INDEX idx_name ON users(name); DROP INDEX idx_name ON users; -- មើល indexes SHOW INDEX FROM users;

12.2 Views

CREATE VIEW active_users AS SELECT id, name, email FROM users WHERE status = 'active'; SELECT * FROM active_users;
📘 ការពន្យល់៖: Index បង្កើនល្បឿន query (ដូច index ក្នុងសៀវភៅ)។ View ជា query ដែលរក្សាទុកជាតារាង virtual — មានប្រយោជន៍សម្រាប់ query ស្មុគស្មាញដដែលៗ។

13. Stored Procedures

13.1 Procedures

DELIMITER // CREATE PROCEDURE GetUsers() BEGIN SELECT * FROM users; END // DELIMITER ; CALL GetUsers();

13.2 Functions

DELIMITER // CREATE FUNCTION DoubleAge(age INT) RETURNS INT DETERMINISTIC BEGIN RETURN age * 2; END // DELIMITER ; SELECT DoubleAge(20); -- 40

13.3 Triggers

CREATE TRIGGER before_insert_user BEFORE INSERT ON users FOR EACH ROW SET NEW.created_at = NOW();
📘 ការពន្យល់៖: Stored Procedure ជាកូដ SQL ដែលរក្សាទុកក្នុង database។ Function return តម្លៃ។ Trigger ដំណើរការស្វ័យប្រវត្តិនៅពេលមាន event (INSERT/UPDATE/DELETE)។

14. Transactions & Security

14.1 Transactions

START TRANSACTION; UPDATE accounts SET balance = balance - 100 WHERE id = 1; UPDATE accounts SET balance = balance + 100 WHERE id = 2; COMMIT; -- រក្សាទុក -- ឬ ROLLBACK; -- បោះបង់

14.2 Users & Privileges

CREATE USER 'app'@'localhost' IDENTIFIED BY 'password'; GRANT SELECT, INSERT ON mydb.* TO 'app'@'localhost'; REVOKE INSERT ON mydb.* FROM 'app'@'localhost'; FLUSH PRIVILEGES;

14.3 Backup

# Backup mysqldump -u root -p mydb > backup.sql # Restore mysql -u root -p mydb < backup.sql
📘 ការពន្យល់៖: Transaction ធានាថាប្រតិបត្តិការទាំងអស់ជោគជ័យ ឬបរាជ័យទាំងអស់ (ACID)។ ត្រូវប្រើសម្រាប់ការផ្ទេរលុយ។ កំណត់ user privileges ឲ្យតិចបំផុតតាមតម្រូវការ។

15. MySQL Project

សាកល្បងបង្កើត Simple Blog Database តូចមួយ៖

-- បង្កើត database CREATE DATABASE blog; USE blog; -- តារាង users CREATE TABLE users ( id INT AUTO_INCREMENT PRIMARY KEY, username VARCHAR(50) UNIQUE NOT NULL, email VARCHAR(100) UNIQUE NOT NULL, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ); -- តារាង posts CREATE TABLE posts ( id INT AUTO_INCREMENT PRIMARY KEY, user_id INT NOT NULL, title VARCHAR(200) NOT NULL, body TEXT, published BOOLEAN DEFAULT FALSE, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE ); -- បញ្ចូលទិន្នន័យ INSERT INTO users (username, email) VALUES ('dara', 'dara@blog.com'), ('sok', 'sok@blog.com'); INSERT INTO posts (user_id, title, body, published) VALUES (1, 'Hello MySQL', 'My first post!', TRUE), (1, 'Draft', 'Not ready', FALSE); -- ស្វែងរក post ដែលបាន publish ជាមួយឈ្មោះអ្នកនិពន្ធ SELECT p.title, u.username FROM posts p JOIN users u ON p.user_id = u.id WHERE p.published = TRUE;
📘 ការពន្យល់៖: គម្រោងនេះប្រើ CREATE TABLE, FOREIGN KEY, INSERT, SELECT, JOIN និង WHERE។ សាកល្បងបន្ថែម comments table, tags, ឬ full-text search។
គន្លឹះ៖: សាកល្បងគម្រោងតូចៗ — Blog, Shop, School System — វានឹងជួយអ្នកយល់ MySQL លឿនជាងអានទ្រឹស្តី។
