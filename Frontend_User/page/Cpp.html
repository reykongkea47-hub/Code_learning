1. C++ Introduction - ការណែនាំ C++

1.1 អ្វីជា C++?

C++ គឺជាភាសាសរសេរកម្មវិធីដ៏មានឥទ្ធិឫទ្ធិមួយ ដែលបង្កើតឡើងដោយ Bjarne Stroustrup ក្នុងឆ្នាំ 1979 នៅ Bell Labs។

📘 ការពន្យល់៖ C++ គឺជាភាសាដែលត្រូវបាន compile (បំប្លែងទៅជា machine code) មុនពេលដំណើរការ។ វាលឿនខ្លាំង ហើយអាចគ្រប់គ្រង memory បានផ្ទាល់។ ប្រើសម្រាប់៖
  • Game Engine — ដូចជា Unreal Engine
  • Operating System — Windows, Linux
  • Browser — Chrome, Firefox
  • Database — MySQL, MongoDB

1.2 ការដំឡើង Compiler

ដើម្បីសរសេរ C++ អ្នកត្រូវការ Compiler និង IDE។

  • Compiler: g++, clang++, MSVC
  • IDE: VS Code, CLion, Dev-C++
📘 ការពន្យល់៖ Compiler គឺជាកម្មវិធីដែលបំប្លែងកូដ C++ ទៅជាភាសាដែលកុំព្យូទ័រយល់។ IDE គឺជាកម្មវិធីសម្រាប់សរសេរ និងរត់កូដ។

1.3 កម្មវិធីដំបូង

#include <iostream> using namespace std; int main() { cout << "Hello, C++!" << endl; return 0; }
📘 ការពន្យល់លម្អិត៖
  • #include <iostream> — បញ្ចូល library សម្រាប់ input/output
  • using namespace std; — អនុញ្ញាតឲ្យប្រើ cout, endl ដោយមិនចាំបាច់សរសេរ std::
  • int main() — ជាចំណុចចាប់ផ្តើមរបស់កម្មវិធី
  • cout << ... — បង្ហាញអត្ថបទលើអេក្រង់
  • return 0; — បញ្ជាក់ថាកម្មវិធីបញ្ចប់ដោយជោគជ័យ

1.4 Comment

// Single-line comment /* Multi-line comment */
📘 ការពន្យល់៖ Comment គឺជាអត្ថបទដែល Compiler មិនអាន ដើម្បីពន្យល់កូដ ឬកត់ត្រាការងារ។ ប្រើ // សម្រាប់មួយបន្ទាត់ និង /* */ សម្រាប់ច្រើនបន្ទាត់។

2. Variables & Data Types

2.1 Variables

int age = 20; double price = 9.99; char grade = 'A'; bool isActive = true;
📘 ការពន្យល់៖ Variable គឺជាប្រអប់សម្រាប់ផ្ទុកទិន្នន័យ។ រាល់ Variable ត្រូវមាន type និង ឈ្មោះ។ ឧទាហរណ៍ int age = 20; មានន័យថា បង្កើត variable ឈ្មោះ age ដែលផ្ទុកលេខ 20។

2.2 Data Types

  • int — ចំនួនគត់ (1, 2, 100, -5)
  • float / double — ចំនួនទសភាគ (3.14, 9.99)
  • char — តួអក្សរមួយ ('A', 'b')
  • bool — ពិត ឬ មិនពិត (true / false)
  • string — អត្ថបទ ("Hello")
📘 ការពន្យល់៖ Data Type កំណត់ថា variable អាចផ្ទុកអ្វី និងប្រើ memory ប៉ុន្មាន។ ឧទាហរណ៍ int ប្រើ 4 bytes ខណៈ char ប្រើ 1 byte។

2.3 Constants

const double PI = 3.14159;
📘 ការពន្យល់៖ const ធ្វើឲ្យ variable មិនអាចផ្លាស់ប្តូរតម្លៃបានទេ។ ប្រើសម្រាប់តម្លៃថេរ ដូចជា PI, gravity ។ល។

2.4 Type Casting

int x = 10; double y = (double)x; // 10.0
📘 ការពន្យល់៖ Type Casting គឺការបំប្លែង type មួយទៅ type មួយទៀត។ ឧទាហរណ៍ បំប្លែង int ទៅ double ដើម្បីទទួលបានចំនួនទសភាគ។

3. Operators

3.1 Arithmetic

int sum = 5 + 3; // 8 int diff = 10 - 4; // 6 int prod = 3 * 4; // 12 int div = 10 / 3; // 3 int mod = 10 % 3; // 1 (សំណល់)
📘 ការពន្យល់៖ Operator ទាំងនេះប្រើសម្រាប់គណនាលេខ។ % គឺ modulo ដែលឲ្យសំណល់នៃការចែក (10 ចែក 3 សល់ 1)។

3.2 Comparison

== // ស្មើ != // មិនស្មើ > // ធំជាង < // តូចជាង >= // ធំជាង ឬស្មើ <= // តូចជាង ឬស្មើ
📘 ការពន្យល់៖ Comparison Operator ត្រឡប់ true ឬ false។ ឧទាហរណ៍ 5 > 3 ត្រឡប់ true។

3.3 Logical

&& // AND — ពិតទាំងពីរ || // OR — ពិតមួយក៏បាន ! // NOT — បញ្ច្រាស
📘 ការពន្យល់៖ Logical Operator ប្រើសម្រាប់ភ្ជាប់ conditions។ ឧទាហរណ៍ (age > 18) && (age < 60) ត្រូវពិតទាំងពីរ។

3.4 Assignment

= // កំណត់តម្លៃ += // បូកបន្ថែម -= // ដកចេញ *= // គុណ /= // ចែក
📘 ការពន្យល់៖ x += 5 ស្មើនឹង x = x + 5។ វាជួយសរសេរកូដខ្លី និងអានងាយ។

4. Control Flow

4.1 if / else

if (age >= 18) { cout << "Adult"; } else { cout << "Minor"; }
📘 ការពន្យល់៖ if ពិនិត្យលក្ខខណ្ឌ។ បើពិត ធ្វើកូដក្នុង { }។ បើមិនពិត ទៅ else។ ឧទាហរណ៍ បើ age >= 18 បង្ហាញ "Adult" បើមិនដូច្នេះ "Minor"។

4.2 switch

switch (day) { case 1: cout << "Mon"; break; case 2: cout << "Tue"; break; default: cout << "?"; }
📘 ការពន្យល់៖ switch ជាជម្រើសជំនួស if-else ច្រើន។ case ជាតម្លៃដែលត្រូវផ្គូផ្គង។ break បញ្ឈប់ការត្រួតពិនិត្យ default ជា fallback។

4.3 for Loop

for (int i = 0; i < 5; i++) { cout << i << " "; }
📘 ការពន្យល់៖ for ប្រើសម្រាប់ធ្វើម្តងទៀតតាមចំនួនជាក់លាក់។ វាមាន 3 ផ្នែក៖ init (i=0), condition (i<5), និង update (i++)។

4.4 while / do-while

int i = 0; while (i < 5) { cout << i; i++; } do { cout << i; i++; } while (i < 5);
📘 ការពន្យល់៖ while ពិនិត្យលក្ខខណ្ឌមុនធ្វើ។ do-while ធ្វើម្តងមុន រួចទើបពិនិត្យ។ ដូច្នេះ do-while ធ្វើយ៉ាងតិចម្តងជាដរាប។

4.5 break / continue

for (int i = 0; i < 10; i++) { if (i == 3) continue; if (i == 7) break; cout << i; }
📘 ការពន្យល់៖ continue រំលង loop ជុំបច្ចុប្បន្ន ហើយបន្តទៅជុំបន្ទាប់។ break បញ្ឈប់ loop ទាំងស្រុង។

5. Functions

5.1 Function Basics

void greet() { cout << "Hello!"; }
📘 ការពន្យល់៖ Function គឺជាកូដដែលអាចហៅឡើងវិញបាន។ void មានន័យថា function នេះមិន return តម្លៃទេ។ ហៅដោយ greet();។

5.2 Parameters & Return

int add(int a, int b) { return a + b; }
📘 ការពន្យល់៖: Function អាចទទួល parameters (a, b) និង return តម្លៃ (a+b)។ ឧទាហរណ៍ add(2, 3) return 5។

5.3 Function Overloading

int add(int a, int b) { return a + b; } double add(double a, double b) { return a + b; }
📘 ការពន្យល់៖: Overloading អនុញ្ញាតឲ្យ function មានឈ្មោះដូចគ្នា ប៉ុន្តែ parameters ខុសគ្នា។ Compiler ជ្រើសរើស version ត្រឹមត្រូវដោយស្វ័យប្រវត្តិ។

5.4 Recursion

int factorial(int n) { if (n <= 1) return 1; return n * factorial(n - 1); }
📘 ការពន្យល់៖: Recursion គឺ function ដែលហៅខ្លួនឯង។ ត្រូវមាន base case ដើម្បីបញ្ឈប់។ ឧទាហរណ៍ factorial(5) = 5×4×3×2×1 = 120។

6. Arrays & Strings

6.1 Arrays

int nums[5] = {1, 2, 3, 4, 5}; cout << nums[0]; // 1
📘 ការពន្យល់៖: Array គឺជាបណ្តុំនៃតម្លៃដែលមាន type ដូចគ្នា។ Index ចាប់ផ្តើមពី 0។ ដូច្នេះ nums[0] គឺតម្លៃទីមួយ។

6.2 Multi-dimensional

int matrix[2][3] = {{1,2,3}, {4,5,6}};
📘 ការពន្យល់៖: Array 2D គឺដូចជា table មួយ ដែលមានជួរដេក និងជួរឈរ។ ប្រើសម្រាប់ matrix, game board ។ល។

6.3 C-Strings

char name[] = "John";
📘 ការពន្យល់៖: C-String គឺ array នៃ char ដែលបញ្ចប់ដោយ '\0'។ ប្រើតិចក្នុង C++ ទំនើប។

6.4 std::string

#include <string> string name = "John"; cout << name.length(); // 4
📘 ការពន្យល់៖: std::string ងាយស្រួលជាង C-String។ អាចប្រើ .length(), .append(), + ជាដើម។

7. Pointers & References

7.1 Pointers

int x = 10; int* ptr = &x; cout << *ptr; // 10
📘 ការពន្យល់៖: Pointer គឺ variable ដែលផ្ទុក address របស់ variable ផ្សេង។ &x យក address, *ptr យកតម្លៃត្រង់ address នោះ។

7.2 References

int x = 10; int& ref = x; ref = 20; // x ក្លាយជា 20
📘 ការពន្យល់៖: Reference គឺជាឈ្មោះផ្សេងរបស់ variable។ ការផ្លាស់ប្តូរ reference ក៏ផ្លាស់ប្តូរ variable ដើមដែរ។

7.3 Dynamic Memory

int* p = new int(5); delete p;
📘 ការពន្យល់៖: new បង្កើត memory ថ្មី (heap) ខណៈ delete បញ្ចេញវាវិញ។ បើភ្លេច delete នឹងកើត memory leak។

8. OOP - Classes

8.1 Classes & Objects

class Student { public: string name; int age; }; Student s1; s1.name = "Dara";
📘 ការពន្យល់៖: Class គឺជាគំរូ (blueprint) សម្រាប់បង្កើត object។ Object គឺជាកូពីនៃ class។ ឧទាហរណ៍ Student ជា class, s1 ជា object។

8.2 Constructors

class Student { public: Student(string n) { name = n; } string name; };
📘 ការពន្យល់៖: Constructor គឺ function ពិសេសដែលហៅដោយស្វ័យប្រវត្តិនៅពេលបង្កើត object។ វាមានឈ្មោះដូច class និងគ្មាន return type។

8.3 Encapsulation

private: int age; public: void setAge(int a) { age = a; }
📘 ការពន្យល់៖: Encapsulation គឺការលាក់ data (private) និងផ្តល់ access តាម methods (public)។ ជួយការពារ data មិនឲ្យកែដោយផ្ទាល់។

8.4 this Pointer

void setAge(int age) { this->age = age; }
📘 ការពន្យល់៖: this គឺ pointer ទៅ object បច្ចុប្បន្ន។ ប្រើដើម្បីបែងចែក parameter និង member variable ដែលមានឈ្មោះដូចគ្នា។

9. Inheritance & Polymorphism

9.1 Inheritance

class Animal { public: void eat() { cout << "Eating"; } }; class Dog : public Animal { public: void bark() { cout << "Woof"; } };
📘 ការពន្យល់៖: Inheritance គឺ class កូន (Dog) ទទួលមរតកពី class មេ (Animal)។ Dog អាចប្រើ eat() បាន ព្រោះវាទទួលពី Animal។

9.2 Polymorphism

អនុញ្ញាតឱ្យ function ដូចគ្នាបង្ហាញឥរិយាបថផ្សេងៗគ្នាអាស្រ័យលើ object។

📘 ការពន្យល់៖: ឧទាហរណ៍ sound() អាចបន្លឺសម្លេងខុសគ្នានៅពេលហៅលើ Dog ឬ Cat។ នេះហៅថា Polymorphism។

9.3 Virtual Functions

class Animal { public: virtual void sound() { cout << "..."; } };
📘 ការពន្យល់៖: virtual ធ្វើឲ្យ function អាច override បាននៅ class កូន។ នេះជាមូលដ្ឋាននៃ Polymorphism។

9.4 Abstract Classes

class Shape { public: virtual void draw() = 0; // pure virtual };
📘 ការពន្យល់៖: Abstract Class គឺ class ដែលមិនអាចបង្កើត object ផ្ទាល់បាន។ វាមាន pure virtual function (= 0) ដែល class កូនត្រូវ implement។

10. Templates & STL

10.1 Templates

template <typename T> T max(T a, T b) { return (a > b) ? a : b; }
📘 ការពន្យល់៖: Template អនុញ្ញាតឲ្យ function ឬ class ដំណើរការជាមួយ type ណាមួយ។ ឧទាហរណ៍ max<int>(2,5) ឬ max<double>(2.5, 3.7) ។

10.2 Vector

#include <vector> vector<int> v = {1, 2, 3}; v.push_back(4); // បន្ថែម 4
📘 ការពន្យល់៖: vector គឺ array ដែលអាចបន្ថែម/ដក element បានយ៉ាងរស់រវើក។ ល្អជាង array ធម្មតា។

10.3 Map

#include <map> map<string, int> ages; ages["Dara"] = 20;
📘 ការពន្យល់៖: map គឺជាបណ្តុំ key-value។ ឧទាហរណ៍ ឈ្មោះ → អាយុ។ ងាយស្រួលរកមើលតាម key។

10.4 Algorithms

#include <algorithm> sort(v.begin(), v.end()); // តម្រៀប
📘 ការពន្យល់៖: STL Algorithms ដូចជា sort, find, reverse ជួយសម្រួលការងារជាមួយ containers។

11. Exception Handling

11.1 try / catch

try { throw 404; } catch (int e) { cout << "Error: " << e; }
📘 ការពន្យល់៖: try គឺ block ដែលអាចមានកំហុស។ throw បោះ error។ catch ចាប់ error នោះដើម្បីដោះស្រាយ។

11.2 throw

ប្រើសម្រាប់បោះ exception នៅពេលមានបញ្ហា (ឧ. ចែកនឹងសូន្យ)។

11.3 Custom Exceptions

class MyError : public exception { const char* what() const noexcept { return "My custom error"; } };
📘 ការពន្យល់៖: អ្នកអាចបង្កើត exception ផ្ទាល់ខ្លួនដោយស្នងពី std::exception។ មានប្រយោជន៍សម្រាប់កំណត់ error ជាក់លាក់។

12. File I/O

12.1 Write to File

#include <fstream> ofstream file("data.txt"); file << "Hello file!"; file.close();
📘 ការពន្យល់៖: ofstream ប្រើសម្រាប់សរសេរទៅ file។ បើក file, សរសេរ, បន្ទាប់មក close()។

12.2 Read from File

ifstream file("data.txt"); string line; while (getline(file, line)) { cout << line; }
📘 ការពន្យល់៖: ifstream ប្រើសម្រាប់អាន file។ getline អានម្តងមួយបន្ទាត់រហូតដល់ចប់។

13. Smart Pointers

13.1 unique_ptr

#include <memory> unique_ptr<int> p = make_unique<int>(5);
📘 ការពន្យល់៖: unique_ptr គឺ pointer ដែលមានម្ចាស់តែម្នាក់។ វាលុប memory ដោយស្វ័យប្រវត្តិនៅពេលបញ្ចប់ scope។ មិនចាំបាច់ delete ទេ។

13.2 shared_ptr

shared_ptr<int> p1 = make_shared<int>(10); shared_ptr<int> p2 = p1; // ចែករំលែក
📘 ការពន្យល់៖: shared_ptr អនុញ្ញាតឲ្យ pointer ច្រើនចែករំលែក object តែមួយ។ Memory ត្រូវលុបនៅពេល pointer ចុងក្រោយបញ្ចប់។

13.3 weak_ptr

weak_ptr<int> wp = p1; // គ្មានម្ចាស់
📘 ការពន្យល់៖: weak_ptr ជា reference ខ្សោយទៅ shared_ptr ដោយមិនបង្កើន reference count។ ប្រើដើម្បីការពារ circular reference។

14. Modern C++

14.1 auto

auto x = 10; // int auto name = string("Dara"); // string
📘 ការពន្យល់៖: auto អនុញ្ញាតឲ្យ Compiler ស្គាល់ type ដោយស្វ័យប្រវត្តិ ដោយផ្អែកលើតម្លៃ។ កូដខ្លី និងអានងាយ។

14.2 Lambda

auto add = [](int a, int b) { return a + b; }; cout << add(2, 3); // 5
📘 ការពន្យល់៖: Lambda គឺ function ខ្លីៗដែលសរសេរនៅនឹងកន្លែង។ ងាយស្រួលសម្រាប់ callback ឬ function តូចៗ។

14.3 Range-based for

vector<int> v = {1, 2, 3}; for (int n : v) { cout << n << " "; }
📘 ការពន្យល់៖: Range-based for ជួយ loop លើ container ដោយមិនចាំបាច់ប្រើ index។ ខ្លី និងសុវត្ថិភាពជាង។

15. Best Practices

  • ប្រើ const នៅពេលអាចធ្វើបាន
  • ប្រើ std::vector ជំនួស array
  • ប្រើ Smart Pointers ជំនួស raw pointers
  • ប្រើ auto នៅពេល type ច្បាស់
  • រក្សា Code ឲ្យអានងាយ
  • Comment តែពេលចាំបាច់
📘 ការពន្យល់៖: ការសរសេរកូដឲ្យមានគុណភាព មិនត្រឹមតែដំណើរការបានទេ តែត្រូវអានងាយ ថែទាំងាយ និងសុវត្ថិភាព។ ឧទាហរណ៍ ប្រើ std::vector ជំនួស array ដើម្បីជៀសវាងបញ្ហា memory។
